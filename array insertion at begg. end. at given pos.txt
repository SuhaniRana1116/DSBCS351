#include <stdio.h>

#define MAX_SIZE 100  // Maximum array size

// Function to display the array
void displayArray(int arr[], int size) {
    if (size == 0) {
        printf("Array is empty.\n");
        return;
    }
    printf("Array elements: ");
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

// Insert at beginning
int insertAtBeginning(int arr[], int *size, int element) {
    if (*size >= MAX_SIZE) {
        printf("Error: Array is full. Cannot insert.\n");
        return 0;
    }
    // Shift elements right
    for (int i = *size; i > 0; i--) {
        arr[i] = arr[i - 1];
    }
    arr[0] = element;
    (*size)++;
    return 1;
}

// Insert at end
int insertAtEnd(int arr[], int *size, int element) {
    if (*size >= MAX_SIZE) {
        printf("Error: Array is full. Cannot insert.\n");
        return 0;
    }
    arr[*size] = element;
    (*size)++;
    return 1;
}

// Insert at given position (1-based index)
int insertAtPosition(int arr[], int *size, int element, int pos) {
    if (*size >= MAX_SIZE) {
        printf("Error: Array is full. Cannot insert.\n");
        return 0;
    }
    if (pos < 1 || pos > *size + 1) {
        printf("Error: Invalid position.\n");
        return 0;
    }
    // Shift elements right from pos-1
    for (int i = *size; i >= pos; i--) {
        arr[i] = arr[i - 1];
    }
    arr[pos - 1] = element;
    (*size)++;
    return 1;
}

int main() {
    int arr[MAX_SIZE];
    int size = 0; // Current number of elements
    int choice, element, pos;

    printf("Enter initial number of elements (max %d): ", MAX_SIZE);
    if (scanf("%d", &size) != 1 || size < 0 || size > MAX_SIZE) {
        printf("Invalid size.\n");
        return 1;
    }

    printf("Enter %d elements: ", size);
    for (int i = 0; i < size; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            printf("Invalid input.\n");
            return 1;
        }
    }

    while (1) {
        printf("\nMenu:\n");
        printf("1. Insert at Beginning\n");
        printf("2. Insert at End\n");
        printf("3. Insert at Given Position\n");
        printf("4. Display Array\n");
        printf("5. Exit\n");
        printf("Enter your choice: ");

        if (scanf("%d", &choice) != 1) {
            printf("Invalid choice.\n");
            return 1;
        }

        switch (choice) {
            case 1:
                printf("Enter element to insert: ");
                scanf("%d", &element);
                insertAtBeginning(arr, &size, element);
                break;
            case 2:
                printf("Enter element to insert: ");
                scanf("%d", &element);
                insertAtEnd(arr, &size, element);
                break;
            case 3:
                printf("Enter element to insert: ");
                scanf("%d", &element);
                printf("Enter position (1 to %d): ", size + 1);
                scanf("%d", &pos);
                insertAtPosition(arr, &size, element, pos);
                break;
            case 4:
                displayArray(arr, size);
                break;
            case 5:
                printf("Exiting...\n");
                return 0;
            default:
                printf("Invalid choice.\n");
        }
    }
}
