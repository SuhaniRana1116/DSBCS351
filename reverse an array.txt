#include <stdio.h>

void reverseArray(int arr[], int left, int right) {
if (left >= right) return;

int temp = arr[left];
arr[left] = arr[right];
arr[right] = temp;

reverseArray(arr, left + 1, right - 1);
}

int main() {
int arr[] = {1, 2, 3, 4, 5};
int n = sizeof(arr) / sizeof(arr[0]);

reverseArray(arr, 0, n - 1);

for (int i = 0; i < n; i++) {
printf("%d ", arr[i]);
}
return 0;
}
